# Cava P0-E: determinism acceptance -- same snapshot, same mods, same tick count, two runs.
# The datapack installed in [pre] freezes the game from the #minecraft:load tag, so the
# number of ticks that elapse before the first RCON command is zero and absolute game
# time is identical across runs. Then /tick sprint N advances exactly N ticks.
#   pwsh -File tools/run-scenario.ps1 -Scenario tools/scenarios/determinism.txt -Name det1
#   pwsh -File tools/run-scenario.ps1 -Scenario tools/scenarios/determinism.txt -Name det2
# compare testbed/hashes/det1-*.json vs det2-*.json (digestPayload) and the .chunks.tsv diffs

restore base-a
datapack

sleep 4

sleep 3
