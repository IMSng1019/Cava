# Cava P0-E: determinism acceptance -- same snapshot, same mods, same tick count, two runs.
#   * the [pre] datapack runs "tick freeze" + "save-off" from the #minecraft:load tag, so
#     game time is pinned at the snapshot value when the tick loop starts (measured: ticks=1);
#   * "/tick sprint N" advances exactly N ticks as fast as possible and returns the game to
#     its previous state (frozen), so the tick count is exact and nothing ticks during the
#     sleeps.  Measured on 2026-09-22: "/tick sprint 200" while UNFROZEN produced +340 ticks
#     (the game keeps running afterwards), and "/tick step 200" while frozen runs in REAL
#     TIME (20 ticks/s), so a 2 s sleep only advanced ~41 ticks.  sprint + freeze is the
#     combination that is both fast and exact.
#   pwsh -File tools/run-scenario.ps1 -Scenario tools/scenarios/determinism.txt -Name det5
#   pwsh -File tools/run-scenario.ps1 -Scenario tools/scenarios/determinism.txt -Name det6
# compare testbed/hashes/det5-*.json vs det6-*.json and the .chunks.tsv diffs
[pre]
restore base-c
datapack
[steps]
ready 300
sleep 3
cmd tick query
cmd tick sprint 200
sleep 6
cmd tick query
cmd save-all flush
sleep 3
hash t200
cmd tick sprint 200
sleep 6
cmd tick query
cmd save-all flush
sleep 3
hash t400
cmd tick sprint 600
sleep 10
cmd tick query
cmd save-all flush
sleep 3
hash t1000
stop
[post]
hash final
