# Cava P0-E: four-bucket window using Carpet's own profiler command (/profile works on a
# dedicated server and prints the vanilla call tree to the console -- no client needed).
[pre]
restore base-a
loadpack 200
[steps]
ready 300
sleep 10
cmd gamerule doDaylightCycle false
cmd time set 18000
cmd difficulty normal
cmd gamerule doMobSpawning false
cmd save-off
cmd player bot01 spawn at 20.5 80 0.5
cmd player bot02 spawn at 14.5 80 14.5
cmd player bot03 spawn at 0.5 80 20.5
cmd player bot04 spawn at -14.5 80 14.5
cmd player bot05 spawn at -20.5 80 0.5
cmd player bot06 spawn at -14.5 80 -14.5
cmd player bot07 spawn at 0.5 80 -20.5
cmd player bot08 spawn at 14.5 80 -14.5
sleep 8
cmd gamemode survival @a
cmd effect give @a minecraft:resistance 3600 4 true
cmd effect give @a minecraft:regeneration 3600 4 true
cmd function cava:spawn_mobs
sleep 25
cmd profile entities
cmd profile 600
sleep 10
cmd profile 600
sleep 10
cmd spark tps
sleep 5
stop
